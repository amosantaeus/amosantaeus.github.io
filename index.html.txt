<!-- ================= METHOD ================= -->
<section id="method" class="marble-sec">
  <div class="wrap">
    <div class="sec-head">
      <span class="label">I · The Method</span>
      <h2 class="sec-title display">Train by evidence, not by feel</h2>
      <p class="sec-lede">Every principle below comes from current hypertrophy research. No secret techniques — the fundamentals, executed consistently for months.</p>
    </div>
    <div class="method-grid">
      <div class="m-card">
        <h3 class="display">Progressive Overload</h3>
        <p>The non-negotiable driver of growth: do slightly more over time — more weight, more reps, or cleaner technique at the same load. We log every session and beat the logbook, not the ego.</p>
        <span class="ev">Foundational principle of all hypertrophy research</span>
      </div>
      <div class="m-card">
        <h3 class="display">Close to Failure</h3>
        <p>Sets end 1–3 reps before failure (RIR 1–3). Hard enough to grow, controlled enough to recover and keep technique sharp — the sweet spot for beginners.</p>
        <span class="ev">Proximity-to-failure meta-analyses (Robinson et al., 2023)</span>
      </div>
      <div class="m-card">
        <h3 class="display">The Right Volume</h3>
        <p>Roughly 10–20 hard sets per muscle per week is where most growth happens. Beginners grow on the lower end — we start there and add sets only when progress stalls.</p>
        <span class="ev">Dose-response volume research (Schoenfeld et al.)</span>
      </div>
      <div class="m-card">
        <h3 class="display">Compounds First</h3>
        <p>Squat, hinge, press, pull. Big lifts train the most muscle per set and build the base everything stacks on. Isolation work fills the gaps — arms, shoulders, calves.</p>
        <span class="ev">Standard beginner programming best practice</span>
      </div>
      <div class="m-card">
        <h3 class="display">5–30 Reps Build Muscle</h3>
        <p>Growth is similar across a wide rep range when sets are close to failure. We use 5–10 for compounds (easier to load) and 8–15+ for isolation (easier on the joints).</p>
        <span class="ev">Load & rep-range meta-analyses (Schoenfeld et al., 2017)</span>
      </div>
      <div class="m-card">
        <h3 class="display">2× per Week per Muscle</h3>
        <p>Hitting each muscle at least twice weekly distributes volume into higher-quality sets and gives you more practice on the lifts. That's why the base program is full-body, 3 days a week.</p>
        <span class="ev">Training frequency research, volume-matched</span>
      </div>
    </div>
  </div>
</section>

      <span class="label">III · Nutrition</span>
      <h2 class="sec-title display">Eat in a surplus. Track it. Repeat.</h2>
      <p class="sec-lede">You cannot out-train an empty plate. Bulking lean means a moderate, measured surplus — enough to build muscle, not enough to build a gut. The targets I set for every beginner:</p>
    </div>
    <div class="nut-grid">
      <div class="nut-facts">
        <div class="fact">
          <div class="fact-v display">+300–500 <span>kcal</span></div>

<script>
  function calc(){
    var bw = parseFloat(document.getElementById('bw').value);
    var act = parseFloat(document.getElementById('act').value);
    if(!bw || bw < 40 || bw > 120){
      document.getElementById('oMaint').textContent = 'Enter 40–120 kg';
      return;
    }
    var maint = Math.round(bw * act / 10) * 10;
    var bulkLo = maint + 300, bulkHi = maint + 500;
    var pLo = Math.round(bw * 1.6), pHi = Math.round(bw * 2.2);
    var gLo = (bw * 0.0025).toFixed(2), gHi = (bw * 0.005).toFixed(2);
    document.getElementById('oMaint').textContent = '~' + maint.toLocaleString() + ' kcal';
    document.getElementById('oBulk').textContent = bulkLo.toLocaleString() + '–' + bulkHi.toLocaleString() + ' kcal';
    document.getElementById('oProt').textContent = pLo + '–' + pHi + ' g/day';
    document.getElementById('oGain').textContent = '+' + gLo + '–' + gHi + ' kg/week';
  }
  document.getElementById('calcBtn').addEventListener('click', calc);
  document.getElementById('bw').addEventListener('keydown', function(e){ if(e.key === 'Enter') calc(); });
  calc();

  function intakeText(){
    function v(id){ var el = document.getElementById(id); return (el.value || '').trim(); }
    var lines = [
      'COACHING APPLICATION — amosantaeus.com',
      '',
      'Name: ' + (v('qName') || '-'),
      'Age: ' + (v('qAge') || '-'),
      'Height: ' + (v('qHeight') ? v('qHeight') + ' cm' : '-'),
      'Weight: ' + (v('qWeight') ? v('qWeight') + ' kg' : '-'),
      'Training experience: ' + (v('qExp') || '-'),
      'Training days/week: ' + (v('qDays') || '-'),
      'Goal: ' + (v('qGoal') || '-'),
      'Injuries / limitations: ' + (v('qInjury') || '-'),
      'Training location: ' + (v('qGym') || '-'),
      'Interested in: ' + (v('qPack') || '-'),
      'Notes: ' + (v('qNotes') || '-')
    ];
    return lines.join('\n');
  }
  function intakeValid(){
    var need = ['qName','qAge','qHeight','qWeight','qExp','qGoal'];
    for(var i=0;i<need.length;i++){
      var el = document.getElementById(need[i]);
      if(!(el.value || '').trim()){
        el.focus();
        document.getElementById('qHint').textContent = 'Please fill in at least name, age, height, weight, experience and goal.';
        return false;
      }
    }
    return true;
  }
  document.getElementById('qMail').addEventListener('click', function(){
    if(!intakeValid()) return;
    var subject = encodeURIComponent('Coaching Application — ' + document.getElementById('qName').value.trim());
    var body = encodeURIComponent(intakeText());
    window.location.href = 'mailto:coaching@amosantaeus.com?subject=' + subject + '&body=' + body;
  });
  document.getElementById('qCopy').addEventListener('click', function(){
    if(!intakeValid()) return;
    var t = intakeText();
    function done(){ document.getElementById('qHint').textContent = 'Copied! Now paste it into a DM to @amosantaeus on Instagram.'; }
    if(navigator.clipboard && navigator.clipboard.writeText){
      navigator.clipboard.writeText(t).then(done, function(){ fallback(); });
    } else { fallback(); }
    function fallback(){
      var ta = document.createElement('textarea');
      ta.value = t; document.body.appendChild(ta); ta.select();
      try{ document.execCommand('copy'); done(); }catch(e){}
      document.body.removeChild(ta);
    }
  });

  // ---- entry gate ----
  document.body.classList.add('locked');
  document.getElementById('gEnter').addEventListener('click', function(){
    function gv(id){ return (document.getElementById(id).value || '').trim(); }

    for(var i=0;i<need.length;i++){
      var el = document.getElementById(need[i]);
      if(!(el.value || '').trim()){
        el.focus();
        document.getElementById('gHint').textContent = 'Please fill in every field.';
        return;
      }
    }
    var h = parseFloat(gv('gHeight')), w = parseFloat(gv('gWeight'));
    var bmi = w / Math.pow(h/100, 2);
    var qualifies = (h >= 180) && (bmi <= 24);   // "tall & skinny" rule — adjust here
    if(qualifies){
      // prefill the application form
      var map = {gName:'qName', gAge:'qAge', gHeight:'qHeight', gWeight:'qWeight', gGoal:'qGoal'};
      for(var k in map){ document.getElementById(map[k]).value = gv(k); }
      var exp = gv('gExp');
      var sel = document.getElementById('qExp');
      for(var j=0;j<sel.options.length;j++){ if(sel.options[j].text === exp){ sel.selectedIndex = j; break; } }
      document.getElementById('gate').classList.add('hidden');
      document.body.classList.remove('locked');
      // prefill calculator too
      if(w >= 40 && w <= 120){ document.getElementById('bw').value = w; calc(); }
    } else {
      document.querySelector('#gate .f-grid').style.display = 'none';
      document.querySelector('#gate .f-actions').style.display = 'none';
      document.getElementById('gHint').style.display = 'none';

    }
  });
